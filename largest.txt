 #include<stdio.h>
 void lgarray()
 {
     int arr[10],i,n;
     int max=0;
     printf("enter limit:");
     scanf("%d",&n);
     printf("enter elements:");
     for(i=0;i<n;i++)
     {
         scanf("%d",&arr[i]);
         if(arr[i]>max){
             max=arr[i];
         }
     }
     printf("greatest =%d ",max);
 }
 int main()
 {
     lgarray();
 }

output:-
enter limit:3
enter elements:2
3
4
greatest =4
